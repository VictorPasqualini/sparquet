# Sparquet Studio local execution bridge.
#
# The framework is a dependency like any other, pinned to the range this Studio
# was built against. The same range is declared in `compat.py`, which is what
# `/health` checks the installed version against, and `test_compat.py` fails if
# the two ever disagree.
#
# Inside the Sparquet repository this line is already satisfied by the editable
# install (`pip install -e ..` from here, or `pip install -e .` at the root), so
# pip leaves it alone and the checkout stays the framework that runs.
sparquet>=0.12,<0.13

# `pyspark` is NOT listed: the framework declares it, and it declares a floor
# (`pyspark>=3.4`) rather than a version, because which Spark line to install is
# the cluster's decision and not this service's.

fastapi>=0.110,<1.0
uvicorn[standard]>=0.27,<1.0
pydantic>=2.6,<3.0

# Encrypts the local secret store. Optional at runtime rather than required: a
# runner without it still boots and still serves secrets whose provider is `env`
# — it refuses only to seal or open a `local` one.
cryptography>=42.0
